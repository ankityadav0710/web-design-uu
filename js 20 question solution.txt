// Define two numbers to compare
const num1 = 15;
const num2 = 10;

console.log(`Comparing ${num1} and ${num2}:\n`);

// 1. Greater than (>)
if (num1 > num2) {
    console.log(`${num1} is greater than ${num2} (num1 > num2 is true)`);
}

// 2. Less than (<)
if (num1 < num2) {
    console.log(`${num1} is less than ${num2} (num1 < num2 is true)`);
}

// 3. Strict Equality (===)
if (num1 === num2) {
    console.log(`${num1} is exactly equal to ${num2} (num1 === num2 is true)`);
}

// 4. Strict Inequality (!==)
if (num1 !== num2) {
    console.log(`${num1} is not equal to ${num2} (num1 !== num2 is true)`);
}

// 5. Greater than or equal to (>=)
if (num1 >= num2) {
    console.log(`${num1} is greater than or equal to ${num2} (num1 >= num2 is true)`);
}

// 6. Less than or equal to (<=)
if (num1 <= num2) {
    console.log(`${num1} is less than or equal to ${num2} (num1 <= num2 is true)`);
}

let hasPermission = false;
let userAge = 20;

// 1. Logical AND (&&) - Returns true if BOTH conditions are true
console.log("--- 1. Logical AND (&&) ---");
console.log("isLoggedIn && hasPermission:", isLoggedIn && hasPermission); // false (true && false)
console.log("isLoggedIn && (userAge >= 18):", isLoggedIn && (userAge >= 18)); // true (true && true)

// 2. Logical OR (||) - Returns true if AT LEAST ONE condition is true
console.log("\n--- 2. Logical OR (||) ---");
console.log("isLoggedIn || hasPermission:", isLoggedIn || hasPermission); // true (true || false)
console.log("hasPermission || (userAge < 18):", hasPermission || (userAge < 18)); // false (false || false)

// 3. Logical NOT (!) - Inverts the boolean value
console.log("\n--- 3. Logical NOT (!) ---");
console.log("!isLoggedIn:", !isLoggedIn); // false (inverts true to false)
console.log("!hasPermission:", !hasPermission); // true (inverts false to true)

// Practical Example with an if-else statement

  console.log("Access granted: User is an adult and logged in.");
} else {
  console.log("Access denied.");

// Function to calculate grade based on marks
function calculateGrade(marks) {
    let grade;

    // Check for invalid marks first
    if (marks < 0 || marks > 100) {
        return "Invalid marks! Please enter a value between 0 and 100.";
    }

    // Determine the grade using else if ladder
    if (marks >= 90) {
        grade = 'A';
    } else if (marks >= 80) {
        grade = 'B';
    } else if (marks >= 70) {
        grade = 'C';
    } else if (marks >= 60) {
        grade = 'D';
    } else {
        grade = 'F'; // Executed if marks are below 60
    }

    return `Marks: ${marks}, Grade: ${grade}`;
}

// Test cases
console.log(calculateGrade(95)); 
console.log(calculateGrade(82));
console.log(calculateGrade(74));
console.log(calculateGrade(61)); 
console.log(calculateGrade(45));
console.log(calculateGrade(105));

function checkLeapYear(year) {
  if ((year % 4 === 0 && year % 100 !== 0) || (year % 400 === 0)) {
    console.log(`${year} is a leap year.`);
  } else {
    console.log(`${year} is not a leap year.`);
  }
}

function sumOfNaturalNumbers(n) {
    if (n < 1) return 0; // Natural numbers start from 1
    return (n * (n + 1)) / 2;
}

// Example usage:
const n = 10;
console.log(`Sum of first ${n} natural numbers is:`, sumOfNaturalNumbers(n));

function calculateFactorial(num) {
    // 1. Handle negative numbers (factorial is not defined)
    if (num < 0) {
        return "Factorial is not defined for negative numbers.";
    }
    
    // 2. Initialize the result variable to 1
    let result = 1;
    
    // 3. Loop from 2 up to the target number
    for (let i = 2; i <= num; i++) {
        result *= i; // Same as result = result * i
    }
    
    return result;
}

// Example usage:
const number = 5;
console.log(`The factorial of ${number} is: ${calculateFactorial(number)}`);
